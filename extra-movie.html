0. 어떤 실습인가

2~4차시의 박스오피스 데이터는 최근 1년 치였습니다. 이번에는 범위를 2004년부터 지금까지 23년치로 넓힙니다. 이 정도면 천만 영화의 등장, 코로나19로 인한 급감, 이후 회복 같은 큰 사건이 데이터에 드러나기에 충분합니다. 절1만 KOBIS 라이브 API를 쓰고(수업에서 쓰던 공용 키 그대로), 절2·절3은 정리된 CSV를 읽으므로 호출 부담이 없습니다.

2. 절1 · 박스오피스 타임머신 (①②③)

크리스마스, 광복절, 어린이날 — 특정 날짜에 해마다 극장에서 무슨 일이 있었는지 10년 전으로 거슬러 올라가 봅니다. 2차시에서 쓴 KOBIS 일별 박스오피스 API를 그대로 쓰되, 이번에는 같은 날짜를 연도만 바꿔 여러 번 조회합니다. 키는 새로 만들 것 없이 포트폴리오 앱의 비밀 금고에 이미 있는 KOBIS_KEY를 그대로 씁니다 — 4차시에 배운 대로 금고는 앱 전체가 공유합니다.

아래에 공식 API 문서에서 복사한 내용을 붙였어.

- 월과 일을 고르면, 최근 10년 동안 해마다 그 날짜의 박스오피스를 조회해서
  연도별 1위 영화를 카드로 죽 보여 줘. 카드에는 연도, 영화 이름, 그날 관객수, 개봉일이 보이게.
- 자주 찾는 날짜는 버튼으로 준비해 줘: 크리스마스(12월 25일), 광복절(8월 15일), 어린이날(5월 5일).
- 카드 아래에는 연도별 1위 영화의 그날 관객수를 세로 막대그래프로 나란히 보여 줘.
  어느 해의 그 날짜가 가장 붐볐는지 한눈에 보이게.
- 연도를 하나 고르면 그 해 그 날짜의 TOP 10 전체 표도 볼 수 있게 해 줘.
- 인증키는 비밀 금고(secrets)의 KOBIS_KEY에서 불러와. 코드에는 절대 쓰지 마.
- 한 번 불러온 날짜의 결과는 다시 부르지 말고 기억해 줘(하루쯤). 10년치면 호출이 10번이니까.
- 올해 날짜가 아직 오지 않았거나 집계 전이면 그 연도 카드는 "집계 전"으로 표시해 줘.
- 숫자가 글자로 오니 숫자로 바꿔서 정렬과 그래프에 써 줘.
- 초보자용 한국어 주석을 달고 페이지 파일 전체 코드를 한 번에 줘.

요청변수: key(발급받은 인증키) · targetDt(조회 날짜, yyyymmdd 여덟 자리)

  ※ 숫자 값이 전부 문자열로 온다.
  ※ 과거 날짜는 넉넉히 열려 있다(10년 전 크리스마스도 그날의 TOP 10이 그대로 온다).

import datetime
import streamlit as st
import pandas as pd
import requests
import plotly.express as px

URL = "https://www.kobis.or.kr/kobisopenapi/webservice/rest/boxoffice/searchDailyBoxOfficeList.json"
올해 = datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=9))).year

st.header("박스오피스 타임머신")

c1, c2, c3 = st.columns(3)
if c1.button("크리스마스"): st.session_state.update(월=12, 일=25)
if c2.button("광복절"): st.session_state.update(월=8, 일=15)
if c3.button("어린이날"): st.session_state.update(월=5, 일=5)
month = st.number_input("월", 1, 12, key="월", value=st.session_state.get("월", 12))
day = st.number_input("일", 1, 31, key="일", value=st.session_state.get("일", 25))


@st.cache_data(ttl=3600)
def fetch(ymd):
    res = requests.get(URL, params={"key": st.secrets["KOBIS_KEY"], "targetDt": ymd}, timeout=20)
    answer = res.json().get("boxOfficeResult")
    return answer["dailyBoxOfficeList"] if answer else []


rows = []
for year in range(올해 - 10, 올해):
    top = fetch(f"{year}{month:02d}{day:02d}")
    if top:
        m = top[0]
        열린날 = pd.to_datetime(m["openDt"], errors="coerce")
        지난날 = (pd.Timestamp(f"{year}-{month:02d}-{day:02d}") - 열린날).days if pd.notna(열린날) else None
        rows.append({"연도": year, "영화명": m["movieNm"], "관객": int(m["audiCnt"]),
                     "개봉일": m["openDt"], "개봉 후": 지난날})

if not rows:
    st.info("그 날짜에는 기록이 없습니다.")
    st.stop()

board = pd.DataFrame(rows)
for _, r in board.iterrows():
    안내 = f"개봉 {int(r['개봉 후'])}일째" if pd.notna(r["개봉 후"]) else "개봉일 모름"
    st.write(f"**{r['연도']}년 — {r['영화명']}** · {r['관객']:,}명 · {안내}")

st.plotly_chart(px.bar(board, x="연도", y="관객", hover_name="영화명"), width="stretch")

고른해 = st.selectbox("그 해 TOP 10 보기", board["연도"])
top10 = fetch(f"{고른해}{month:02d}{day:02d}")
st.dataframe(pd.DataFrame([{"순위": int(m["rank"]), "영화명": m["movieNm"],
                            "관객": int(m["audiCnt"])} for m in top10]), hide_index=True)
st.caption(f"{month}월 {day}일 · 최근 10년")

import datetime

import pandas as pd
import plotly.express as px
import requests
import streamlit as st

st.set_page_config(page_title="박스오피스 타임머신", page_icon="🕰️", layout="wide")
st.title("🕰️ 박스오피스 타임머신")
st.caption("월과 일을 고르면, 최근 10년 동안 해마다 그 날짜의 박스오피스 1위를 보여 줍니다.")

# 인증키는 비밀 금고(secrets)에서 불러온다 — 코드에 직접 쓰지 않는다
API_KEY = st.secrets["KOBIS_KEY"]
URL = "https://www.kobis.or.kr/kobisopenapi/webservice/rest/boxoffice/searchDailyBoxOfficeList.json"

# 오늘 날짜는 한국 시간 기준으로 계산한다
KST = datetime.timezone(datetime.timedelta(hours=9))
today = datetime.datetime.now(KST).date()

# ── 날짜 고르기: 자주 찾는 날짜는 버튼으로 ──
if "month" not in st.session_state:
    st.session_state.month = 12  # 처음 열면 크리스마스
    st.session_state.day = 25


def set_date(m, d):
    """프리셋 버튼을 누르면 월과 일을 바꾼다."""
    st.session_state.month = m
    st.session_state.day = d


b1, b2, b3 = st.columns(3)
b1.button("🎄 크리스마스 (12월 25일)", on_click=set_date, args=(12, 25))
b2.button("🇰🇷 광복절 (8월 15일)", on_click=set_date, args=(8, 15))
b3.button("🎈 어린이날 (5월 5일)", on_click=set_date, args=(5, 5))

c1, c2 = st.columns(2)
month = c1.number_input("월", min_value=1, max_value=12, key="month")
day = c2.number_input("일", min_value=1, max_value=31, key="day")


def date_or_none(year, m, d):
    """달력에 없는 날짜(2월 30일 등)면 None을 돌려준다."""
    try:
        return datetime.date(year, m, d)
    except ValueError:
        return None


this_date = date_or_none(today.year, month, day)
if this_date is None:
    st.error("달력에 없는 날짜입니다. 월과 일을 다시 골라 주세요.")
    st.stop()

# 올해 그 날짜가 아직 오지 않았으면 올해 카드는 "집계 전"으로 두고,
# 데이터가 있는 해가 10년이 되도록 한 해 더 거슬러 간다
if this_date >= today:
    years = list(range(today.year - 10, today.year + 1))
else:
    years = list(range(today.year - 9, today.year + 1))


@st.cache_data(ttl=86400)  # 한 번 불러온 날짜의 결과는 하루쯤 기억해 둔다
def fetch_boxoffice(date_str):
    """해당 날짜의 일별 박스오피스 TOP 10 목록을 받아 온다."""
    params = {"key": API_KEY, "targetDt": date_str}
    res = requests.get(URL, params=params, timeout=10)
    res.raise_for_status()
    data = res.json()
    # 인증키가 틀리면 상태코드 200에 faultInfo 상자가 온다
    if "faultInfo" in data:
        raise RuntimeError(data["faultInfo"].get("message", "API 오류"))
    return data.get("boxOfficeResult", {}).get("dailyBoxOfficeList", [])


st.header(f"{month}월 {day}일, 해마다 극장에서는")

records = []  # 연도별 1위를 모아 두는 목록 (막대그래프용)
tables = {}  # 연도별 TOP 10 전체 표

shown = list(reversed(years))  # 최근 해부터 보여 준다
for i in range(0, len(shown), 4):
    row = st.columns(4)
    for col, year in zip(row, shown[i : i + 4]):
        target = date_or_none(year, month, day)
        with col:
            st.markdown(f"#### {year}년")
            # 아직 오지 않은 날짜(올해)는 집계 전으로 표시한다
            if target is None or target >= today:
                st.markdown("🕐 집계 전")
                continue
            try:
                movies = fetch_boxoffice(target.strftime("%Y%m%d"))
            except Exception:
                st.markdown("불러오지 못했습니다 — 잠시 뒤 새로고침해 주세요")
                continue
            if not movies:
                st.markdown("🕐 집계 전")
                continue
            df = pd.DataFrame(movies)
            # 숫자가 글자로 오므로 숫자로 바꿔야 정렬과 그래프에 쓸 수 있다
            for c in ["rank", "audiCnt", "audiAcc"]:
                df[c] = pd.to_numeric(df[c])
            tables[year] = df
            top = df.sort_values("rank").iloc[0]
            st.markdown(f"**{top['movieNm']}**")
            st.markdown(f"그날 관객 {top['audiCnt']:,}명")
            st.markdown(f"개봉일 {top['openDt']}")
            records.append(
                {"연도": str(year), "영화명": top["movieNm"], "그날 관객수": top["audiCnt"]}
            )

if records:
    # ── 연도별 1위 영화의 그날 관객수 막대그래프 ──
    st.subheader("연도별 1위 영화의 그날 관객수")
    chart_df = pd.DataFrame(records).sort_values("연도")
    fig = px.bar(chart_df, x="연도", y="그날 관객수", hover_data=["영화명"])
    st.plotly_chart(fig, width="stretch")

    # ── 연도를 하나 고르면 그 해 그 날짜의 TOP 10 전체 표 ──
    st.subheader("그 해 그 날짜의 TOP 10 전체 보기")
    pick = st.selectbox("연도 고르기", sorted(tables.keys(), reverse=True))
